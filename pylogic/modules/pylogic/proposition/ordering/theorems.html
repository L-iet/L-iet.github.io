def order_axiom_bf(x, y) -> Implies[Equals, Not[LessThan]]:
    return Implies(
        Equals(x, y),
        Not(LessThan(x, y)),
        _is_proven=True,
        _assumptions=set(),
        _inference=Inference(None, rule="order_axiom_bf"),
    )

def absolute_value_nonnegative_f(x: Term) -> Or[GreaterThan, Equals]:
    """
    Logical inference rule. If x is a real number, returns a proven proposition of the form Abs(x) > 0 V Abs(x) = 0.
    """
    abs_x = Abs(x)
    return Or(
        GreaterThan(abs_x, 0),
        Equals(abs_x, 0),
        _is_proven=True,
        _assumptions=set(),
        _inference=Inference(None, rule="absolute_value_nonnegative_f"),
    )
